#include <metal_stdlib>
using namespace metal;
#pragma METAL fp contract(off)
// SIMD program tile: i belongs to lane i % 32, private slot i / 32.
// Every collective has uniform participation, including padded tail lanes.
// FP32 local-slot accumulation then SIMD sum/max; precise rsqrt; fast math disabled.
// Peak live lane-owned Buffer values: 10 FP32; temporaries/spills unmodeled.
kernel void cake_rmsnorm(
    device const float* v0 [[buffer(0)]],
    device const float* v1 [[buffer(1)]],
    device float* v2 [[buffer(2)]],
    uint3 program [[threadgroup_position_in_grid]],
    uint lane [[thread_index_in_simdgroup]]) {
    float v3[3] = {};
    float v4[3] = {};
    float v5[3] = {};
    float v6[1] = {};
    float v7[1] = {};
    float v8[1] = {};
    float v9[1] = {};
    float v10[3] = {};
    float v11[3] = {};
    // CAKE_OP: load_x
    {
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            v3[s] = i < 65u ? v0[(ulong(program.x)) * 65ul + ((ulong(((i / 1u) % 65u)) + 0ul)) * 1ul] : 0.0f;
        }
    }
    // CAKE_OP: load_weight
    {
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            v4[s] = i < 65u ? v1[((ulong(((i / 1u) % 65u)) + 0ul)) * 1ul] : 0.0f;
        }
    }
    // CAKE_OP: square
    {
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            v5[s] = i < 65u ? (v3[s] * v3[s]) : 0.0f;
        }
    }
    // CAKE_OP: sum
    {
        for (uint output = 0u; output < 1u; ++output) {
            float partial = 0.0f;
            uint begin = (output / 1u) * 65u;
            uint end = begin + 65u;
            for (uint s = begin / 32u; s < (end + 31u) / 32u; ++s) {
                uint i = s * 32u + lane;
                if (i >= begin && i < end && i % 1u == output % 1u) {
                    partial = partial + v5[s];
                }
            }
            float reduced = simd_sum(partial);
            if (lane == output % 32u) v6[output / 32u] = reduced;
        }
    }
    // CAKE_OP: mean
    {
        for (uint s = 0u; s < 1u; ++s) {
            uint i = s * 32u + lane;
            v7[s] = i < 1u ? (v6[s] / 65.0f) : 0.0f;
        }
    }
    // CAKE_OP: variance
    {
        for (uint s = 0u; s < 1u; ++s) {
            uint i = s * 32u + lane;
            v8[s] = i < 1u ? (v7[s] + 1e-05f) : 0.0f;
        }
    }
    // CAKE_OP: rsqrt
    {
        for (uint s = 0u; s < 1u; ++s) {
            uint i = s * 32u + lane;
            v9[s] = i < 1u ? precise::rsqrt(v8[s]) : 0.0f;
        }
    }
    // CAKE_OP: normalized
    {
        float scalar1 = simd_broadcast(v9[0], 0u);
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            v10[s] = i < 65u ? (v3[s] * scalar1) : 0.0f;
        }
    }
    // CAKE_OP: result
    {
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            v11[s] = i < 65u ? (v10[s] * v4[s]) : 0.0f;
        }
    }
    // CAKE_OP: store_out
    {
        for (uint s = 0u; s < 3u; ++s) {
            uint i = s * 32u + lane;
            if (i < 65u) v2[(ulong(program.x)) * 65ul + ((ulong(((i / 1u) % 65u)) + 0ul)) * 1ul] = v11[s];
        }
    }
    // CAKE_KERNEL_END
}
